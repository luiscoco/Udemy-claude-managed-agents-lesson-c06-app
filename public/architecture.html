Step 0 · Prepare the application before the workflow

Use your own Anthropic API key and an existing agent ID and environment ID.

For the optional live path, set your API key, then run npm run setup if you do not already have resource IDs. The script creates the environment before a text-only agent, saves each ID immediately, and reuses IDs already configured. It uses CLAUDE_MODEL or the source default; choose a model available to your account. An uncertain network failure may have created a resource: inspect your account before retrying. Fixtures work without credentials. Set LIVE_ENABLED=true only when you intend to select the live provider.

/** Optional one-time provisioning. Reuses IDs already present in .env. */
import Anthropic from "@anthropic-ai/sdk";
import { existsSync, readFileSync, writeFileSync } from "node:fs";
if (
  !process.env.ANTHROPIC_API_KEY ||
  process.env.ANTHROPIC_API_KEY.startsWith("your_")
) {
  throw new Error("Set your own ANTHROPIC_API_KEY in .env first.");
}
const client = new Anthropic({ maxRetries: 0 });
let contents = existsSync(".env") ? readFileSync(".env", "utf8") : "";
function save(key: string, value: string) {
  const pattern = new RegExp("^\\s*" + key + "=.*$", "m");
  contents = pattern.test(contents)
    ? contents.replace(pattern, key + "=" + value)
    : contents.trimEnd() + "\n" + key + "=" + value + "\n";
  writeFileSync(".env", contents, { mode: 0o600 });
}
const valid = (value: string | undefined) =>
  Boolean(value && !value.startsWith("your_"));
if (!valid(process.env.CLAUDE_MANAGED_ENVIRONMENT_ID)) {
  const environment = await client.beta.environments.create({
    name: "C06 student environment",
    config: { type: "cloud", networking: { type: "limited" } },
  });
  // Save immediately so a later agent failure does not lose the environment ID.
  save("CLAUDE_MANAGED_ENVIRONMENT_ID", environment.id);
}
if (!valid(process.env.CLAUDE_MANAGED_AGENT_ID)) {
  const agent = await client.beta.agents.create({
    name: "C06 text-only student agent",
    model: process.env.CLAUDE_MODEL || "claude-opus-5-5",
    system: "Answer the student's question clearly and briefly.",
    tools: [],
    mcp_servers: [],
    skills: [],
  });
  save("CLAUDE_MANAGED_AGENT_ID", agent.id);
}
console.log(
  "Prerequisite IDs are saved in your local .env. Existing configured IDs were reused.",
);

3 · Understand the lesson-specific rules · part 1

Offers authored completed/incomplete fixtures and an explicitly enabled real provider path. Each run creates a new session; the sample does not offer a multi-turn resume UI.

Source: src/runner.ts · starting at line 1

import Anthropic from "@anthropic-ai/sdk";
export type Frame = { type: string; text?: string; reason?: string };
export interface Provider {
  source: "fixture" | "live";
  create(): Promise<string>;
  open(id: string): Promise<AsyncIterable<Frame>>;
  send(id: string, prompt: string): Promise<void>;
}
export function fixture(incomplete = false): Provider {
  return {
    source: "fixture",
    async create() {
      return "fixture-session";
    },
    async open() {
      return (async function* () {
        yield { type: "running" };
        yield {
          type: "message",
          text: "This is an authored example, not a live model response.",
        };
        if (!incomplete) yield { type: "idle", reason: "end_turn" };
      })();
    },
    async send() {},
  };
}
export function live(client: Anthropic, agent: string, environment: string): Provider {
  let stream: Awaited<ReturnType<typeof client.beta.sessions.events.stream>>;
  return {
    source: "live",
    async create() {
      const session = await client.beta.sessions.create({
        agent,
        environment_id: environment,
        budget: { type: "limit", max_list_cost: { amount: "50", currency: "USD" } },
      });
      return session.id;
    },
    async open(id) {
      stream = await client.beta.sessions.events.stream(id);
      return (async function* () {
        for await (const event of stream) {
          if (event.type === "agent.message") {
            for (const block of event.content)
              if (block.type === "text") yield { type: "message", text: block.text };
          } else if (event.type === "session.status_idle") {
            yield { type: "idle", reason: event.stop_reason.type };
            break;
          } else if (event.type === "session.status_terminated") {
            yield { type: "terminated" };
            break;
          } else if (event.type === "session.error") {
            yield { type: "error", text: "Provider reported a session error." };
            break;
          } else yield { type: event.type };
        }
      })();
    },
    async send(id, prompt) {
      await client.beta.sessions.events.send(id, {
        events: [{ type: "user.message", content: [{ type: "text", text: prompt }] }],
      });
    },
  };
}
export async function run(
  provider: Provider,
  prompt: string,
  emit: (frame: Frame & { source: string }) => void,
) {
  const send = (frame: Frame) => emit({ ...frame, source: provider.source });
  const id = await provider.create();
  send({ type: "session", text: id });
  const stream = await provider.open(id);
  await provider.send(id, prompt);
  let outcome = "incomplete";
  for await (const frame of stream) {
    send(frame);
    if (frame.type === "idle")
      outcome = frame.reason === "end_turn" ? "completed" : `paused: ${frame.reason}`;
    else if (frame.type === "terminated") outcome = "terminated";
    else if (frame.type === "error") outcome = "error";
  }
  send({ type: "outcome", text: outcome });
  return outcome;
}
